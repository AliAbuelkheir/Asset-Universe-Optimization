\newcommand{\limitationcard}[4]{%
  \node[card] (#1) at (#2,0) {};
  \draw[accent]
    ($(#1.north)+(-0.35cm,0)$) -- ($(#1.north)+(0.35cm,0)$);
  \node[cardhead] at ($(#1.north)+(0,-0.7cm)$) {#3};
  \node[cardbody] at ($(#1.center)+(0,-0.35cm)$) {#4};
}
