flowchart LR
    subgraph INPUTS["Inputs"]
        A["Market price, yield, volume, CPI, and USD data"]
        B["Canonical monthly state panel"]
        C["Future interface input: month and risk tolerance"]
    end

    subgraph MODEL["Current ranked-risk model"]
        P["PPO shared scorer"]
        S["Predicted risk score per active asset"]
        R["Monthly reward: rank correlation plus MSE term"]
    end

    subgraph OUTPUT["Current output"]
        O["Ranked active asset universe"]
    end

    subgraph FUTURE["Deferred interface stages"]
        F1["Investor-facing selection layer"]
        F2["Optional allocation model or REST adapter"]
        F3["Optional simulated earnings"]
    end

    A --> B
    B --> P
    P --> S
    S --> O
    R -.-> P
    C -.-> F1
    O -.-> F1
    F1 -.-> F2
    F2 -.-> F3
